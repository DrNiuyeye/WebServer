# 目录导航
pwd                      # 显示当前工作路径
ls                      # 列出当前目录文件
ls -la                  # 列出所有文件（包括隐藏文件）详细信息
cd /home                # 切换到指定目录
cd ~                    # 切换到home目录
cd -                    # 切换到上一个目录

# 目录操作
mkdir dirname           # 创建目录
mkdir -p a/b/c          # 递归创建多级目录
rmdir dirname           # 删除空目录
rm -rf dirname          # 强制删除非空目录

# 文件管理
touch file.txt          # 创建空文件或更新文件时间
cp source dest          # 复制文件
cp -r dir1 dir2         # 递归复制目录
mv old new              # 移动或重命名文件
rm file.txt             # 删除文件

# 文件查看
cat file.txt            # 显示文件内容
less file.txt           # 分页查看文件
head -n 10 file.txt     # 查看前10行
tail -n 20 file.txt     # 查看后20行
tail -f log.log         # 实时查看日志文件

# 系统状态
df -h                   # 查看磁盘使用情况
du -sh *                # 查看当前目录各文件大小
free -h                 # 查看内存使用情况
top                     # 查看进程实时状态
ps aux                  # 查看所有进程
uname -a                # 查看系统信息

# 用户管理
whoami                  # 查看当前用户名
who                     # 查看登录用户
passwd                  # 修改密码

ping host               # 测试网络连接
ifconfig                # 查看网络配置
ip addr                 # 查看IP地址（新版）
netstat -tlnp          # 查看端口占用
wget url                # 下载文件
curl url                # 发送HTTP请求

# tar
tar -czf archive.tar.gz dir/    # 压缩
tar -xzf archive.tar.gz         # 解压

# zip
zip -r archive.zip dir/         # 压缩
unzip archive.zip               # 解压
